# the volume overlay: the remote's volume keys reach the fake cadenced, which steps its level by 5 and sends the level

wait-for route:home

wait 2000
send {"route": {"route": "home"}}
wait-for route:home
wait 800
focus

press VOLUME_UP
wait 60

wait 400
screenshot sim-out/volume-02-up-35.png
press VOLUME_UP
press VOLUME_UP
wait 400
screenshot sim-out/volume-03-up-45.png

focus
