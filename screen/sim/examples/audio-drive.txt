# the drive scenes under the audio page: a cast with its cover, real music streamed into the visualizer feed, the
# title in the shape while lean, the controls staggering in and out. the track is any f32 stereo 48 khz file
# (ffmpeg -i song.mp3 -ac 2 -ar 48000 -f f32le /tmp/song.f32); it stays local.
# `build/screen-sim/sim run screen/sim/examples/audio-drive.txt`; screenshots land in sim-out/.

send {"capability_config": {"capability": "display.screen", "json": {"visualizer": "drive"}}}

pcm-file /tmp/song.f32
wait 3000
screenshot sim-out/audio-drive-01-lean.png
wait 533
screenshot sim-out/audio-drive-02-lean.png
press DOWN
wait 120
